// ===== MODAL =====
let currentStep = 1;
const totalSteps = 3;

function openModal() {
  document.getElementById('modal-overlay').classList.add('open');
  document.body.style.overflow = 'hidden';
  goToStep(1);
}

function closeModal() {
  document.getElementById('modal-overlay').classList.remove('open');
  document.body.style.overflow = '';
}

function handleOverlayClick(e) {
  if (e.target === document.getElementById('modal-overlay')) closeModal();
}

function goToStep(step) {
  currentStep = step;
  for (let i = 1; i <= totalSteps; i++) {
    const stepEl = document.getElementById(`modal-step-${i}`);
    stepEl.classList.toggle('active', i === step);

    const circle = document.getElementById(`step-circle-${i}`);
    circle.classList.remove('active', 'done');
    if (i < step) { circle.classList.add('done'); circle.textContent = '✓'; }
    else if (i === step) { circle.classList.add('active'); circle.textContent = i; }
    else { circle.textContent = i; }

    if (i < totalSteps) {
      const line = document.getElementById(`step-line-${i}`);
      if (line) line.classList.toggle('done', i < step);
    }
  }

  const btnPrev = document.getElementById('btn-prev');
  const btnNext = document.getElementById('btn-next');
  const btnSubmit = document.getElementById('btn-submit');

  btnPrev.style.display = step > 1 ? 'block' : 'none';
  btnNext.style.display = step < totalSteps ? 'block' : 'none';
  btnSubmit.style.display = step === totalSteps ? 'block' : 'none';
}

function stepNext() {
  if (currentStep === 1) {
    const desc = document.getElementById('modal-desc').value.trim();
    if (!desc) {
      document.getElementById('modal-desc').style.borderColor = '#EF4444';
      document.getElementById('modal-desc').focus();
      setTimeout(() => document.getElementById('modal-desc').style.borderColor = '', 2000);
      return;
    }
  }
  if (currentStep < totalSteps) goToStep(currentStep + 1);
}

function stepPrev() {
  if (currentStep > 1) goToStep(currentStep - 1);
}

function submitModal() {
  closeModal();
  alert('Votre demande a été publiée avec succès ! 🎉');
}

function selectCat(el) {
  document.querySelectorAll('.cat-item').forEach(c => c.classList.remove('selected'));
  el.classList.add('selected');
}

function toggleUrgence() {
  const row = document.getElementById('urgence-row');
  const checked = document.getElementById('toggle-urgence').checked;
  row.style.borderColor = checked ? '#EF4444' : 'var(--border)';
  row.style.background = checked ? '#FEF2F2' : '';
}

// ===== TOGGLE CLIENT / PRESTATAIRE =====
function switchView(mode) {
  const sidebarClient = document.getElementById('sidebar-client');
  const sidebarPresta = document.getElementById('sidebar-presta');
  const pillClient = document.getElementById('pill-client');
  const pillPresta = document.getElementById('pill-presta');
  const greetingText = document.getElementById('greeting-text');
  const greetingSub = document.getElementById('greeting-sub');

  sidebarClient.style.display = mode === 'client' ? 'block' : 'none';
  sidebarPresta.style.display = mode === 'presta' ? 'block' : 'none';

  if (mode === 'client') {
    pillClient.className = 'toggle-pill active-client';
    pillPresta.className = 'toggle-pill';
    greetingText.childNodes[0].textContent = 'Bonjour Yassine 👋 ';
    greetingSub.textContent = '';
  } else {
    pillClient.className = 'toggle-pill';
    pillPresta.className = 'toggle-pill active-presta';
    greetingText.childNodes[0].textContent = 'Bonjour Ahmed 👋 ';
    greetingSub.textContent = '· Vos matchs du jour : 8 nouvelles demandes';
  }

  sidebarClient.style.animation = 'none';
  sidebarPresta.style.animation = 'none';
  setTimeout(() => {
    sidebarClient.style.animation = '';
    sidebarPresta.style.animation = '';
  }, 10);
}

// ===== FILTRES CHIPS =====
document.querySelectorAll('.chip').forEach(chip => {
  chip.addEventListener('click', () => {
    document.querySelectorAll('.chip').forEach(c => c.classList.remove('active'));
    chip.classList.add('active');

    const filter = chip.dataset.filter;
    const allCards = document.querySelectorAll('#feed-container [data-categorie]');
    allCards.forEach(card => {
      if (filter === 'tous' || card.dataset.categorie === filter) {
        card.style.display = '';
      } else {
        card.style.display = 'none';
      }
    });
  });
});

// ===== TOGGLE J'AIME =====
function toggleLike(btn) {
  const isLiked = btn.classList.toggle('liked');
  const countEl = btn.querySelector('.like-count');
  const svg = btn.querySelector('svg');
  if (isLiked) {
    svg.style.fill = 'var(--orange)';
    svg.style.stroke = 'var(--orange)';
    countEl.textContent = 'J\'aimé ♥';
    btn.style.color = 'var(--orange)';
  } else {
    svg.style.fill = 'none';
    svg.style.stroke = 'currentColor';
    countEl.textContent = 'J\'aime';
    btn.style.color = '';
  }
}

// ===== TOGGLE FAVORIS PRESTATAIRE =====
function toggleFavoris(btn) {
  btn.classList.toggle('active');
  if (btn.classList.contains('active')) {
    btn.textContent = '♥ Favoris';
  } else {
    btn.textContent = '♡ Favoris';
  }
}

function loadMore() {
  const feed = document.getElementById('feed-container');
  const currentFilter = document.querySelector('.chip.active')?.dataset.filter || 'tous';

  moreCardsData.forEach(d => {
    const html = d.type === 'client' ? buildClientCard(d) : buildPrestaCard(d);
    const wrapper = document.createElement('div');
    wrapper.innerHTML = html.trim();
    const card = wrapper.firstChild;
    feed.appendChild(card);

    if (currentFilter !== 'tous' && card.dataset.categorie !== currentFilter) {
      card.style.display = 'none';
    }
  });

  loadMoreCount++;
  const btn = document.getElementById('btn-load-more');
  if (loadMoreCount >= 2) {
    btn.textContent = 'Plus aucune annonce pour le moment';
    btn.disabled = true;
    btn.style.opacity = '0.5';
    btn.style.cursor = 'default';
  }
}
